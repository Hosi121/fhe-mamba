cmake_minimum_required(VERSION 3.25.2)
project(packed_diagnostics LANGUAGES CXX)
find_package(fideslib REQUIRED CONFIG)
find_package(OpenFHE REQUIRED CONFIG)
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
set(FHE_STAGE0_GPU_RNS ON CACHE BOOL "GPU plaintext preparation" FORCE)

  stage0_layout stage1_mamba2_plan fideslib_plaintext_encoder Threads::Threads OpenMP::OpenMP_CXX)

set_property(TARGET packed_operation_replay PROPERTY COMPILE_FLAGS "${FIDESLIB_CXX_FLAGS}")
