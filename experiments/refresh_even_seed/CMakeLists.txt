cmake_minimum_required(VERSION 3.25.2)
project(refresh_even_seed LANGUAGES CXX CUDA)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(REFRESH_CANDIDATE 0 CACHE STRING "0 for baseline library; 1 for patched library")
if(NOT REFRESH_CANDIDATE MATCHES "^[01]$")
  message(FATAL_ERROR "REFRESH_CANDIDATE must be 0 or 1")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 17)
find_package(fideslib REQUIRED CONFIG)
find_package(OpenFHE REQUIRED CONFIG)
find_package(CUDAToolkit REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${FIDESLIB_CXX_FLAGS}")
set(NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../native/fideslib_stage0/src")
add_executable(refresh_probe probe.cpp ${NATIVE_SOURCE}/fideslib_ring_map.cu)
target_link_libraries(refresh_probe PRIVATE fideslib::fideslib CUDA::cudart)
target_include_directories(refresh_probe PRIVATE ${NATIVE_SOURCE} ${OpenFHE_INCLUDE}
  ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe
  ${OpenFHE_INCLUDE}/third-party/include)

target_include_directories(refresh_probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(refresh_probe PRIVATE REFRESH_CANDIDATE=${REFRESH_CANDIDATE})
